import re
from datetime import datetime, timedelta

def parse_weather_question(question):
    """
    Parse a natural language weather question.

    Args:
        question (str): User's weather-related question

    Returns:
        dict: Extracted information: location, date/time_period, and weather attribute
    """
    question = question.lower()

    # Weather attribute keywords
    attributes = {
        'temperature': ['temperature', 'temp', 'hot', 'cold'],
        'rain': ['rain', 'raining', 'precipitation', 'wet'],
        'humidity': ['humidity', 'humid'],
        'wind': ['wind', 'windy']
    }

    # Extract attribute
    attribute = 'general'  # fallback
    for key, words in attributes.items():
        if any(word in question for word in words):
            attribute = key
            break

    # Extract relative date phrases
    today = datetime.now().date()
    if "today" in question:
        time_period = today.isoformat()
    elif "tomorrow" in question:
        time_period = (today + timedelta(days=1)).isoformat()
    elif "weekend" in question:
        # Assume Sat-Sun of this week
        days_ahead = (5 - today.weekday()) % 7
        saturday = today + timedelta(days=days_ahead)
        sunday = saturday + timedelta(days=1)
        time_period = f"{saturday.isoformat()} to {sunday.isoformat()}"
    else:
        # Look for specific day of week
        days = ['monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday', 'sunday']
        for i, day in enumerate(days):
            if day in question:
                day_num = (i - today.weekday()) % 7
                target_date = today + timedelta(days=day_num)
                time_period = target_date.isoformat()
                break
        else:
            time_period = 'unspecified'

    # Extract location (very simple: assume after "in <location>")
    match = re.search(r'in ([a-z\s]+)', question)
    location = match.group(1).strip().title() if match else None

    return {
        'location': location,
        'time_period': time_period,
        'attribute': attribute
    }
