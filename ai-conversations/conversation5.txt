def parse_weather_question(question):
    """
    A simple mock parser for demonstration.

    """
    # For demo purposes, extract location and date with simple logic
    question = question.lower()
    location = None
    time_period = None
    attribute = None

    # Very naive location detection: check for 'perth' or 'london' as examples

        if loc in question:

    # Detect 'today' or date in question
    if 'today' in question:
        time_period = datetime.now().date().isoformat()
    else:
        import re
        date_match = re.search(r'\d{4}-\d{2}-\d{2}', question)
        if date_match:
            time_period = date_match.group(0)

    # Detect attribute keywords
    if 'temperature' in question or 'temp' in question:
        attribute = 'temperature'
    elif 'rain' in question or 'raining' in question:
        attribute = 'rain'
    elif 'humidity' in question:
        attribute = 'humidity'
    elif 'wind' in question:
        attribute = 'wind'
    else:
        attribute = 'general'

    return {'location': location, 'time_period': time_period, 'attribute': attribute}
